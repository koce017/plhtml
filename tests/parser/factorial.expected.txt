"Factorial"
Main:
    string quit
    While quit NeqOp "q"
        integer n
        integer i
        integer fact
        i = 1
        fact = 1
        Write: "n: "
        Read n
        While i LeqOp n
            fact = fact * i
            i = i + 1
        Write: "Factorial of " + n + " is " + fact
        Write: "\nType q to quit: "
        Read quit
