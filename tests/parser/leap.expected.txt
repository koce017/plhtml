"Leap Year"
Main:
    string quit
    While quit NeqOp "q"
        integer year
        boolean answer
        Write: "year: "
        Read year
        If year % 4 EqOp 0
            If year % 100 NeqOp 0
                answer = true
            If year % 400 EqOp 0
                answer = true
        If answer
            Write: year + " is leap year"
        If !answer
            Write: year + " is not leap year"
        Write: "\nType q to quit: "
        Read quit
