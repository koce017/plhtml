"Fibonacci Series"
Main:
    string quit
    While quit NeqOp "q"
        integer a
        integer b
        integer c
        integer i
        integer n
        a = 0
        b = 1
        i = 1
        Write: "n: "
        Read n
        While i LeqOp n
            Write: a + " "
            c = a + b
            a = b
            b = c
            i = i + 1
        Write: "\nType q to quit: "
        Read quit
